cmake_minimum_required(VERSION 3.16)
project(storvault_agent CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(SV_TSAN "Build with ThreadSanitizer" OFF)
option(SV_ASAN "Build with AddressSanitizer + UBSan" OFF)
add_compile_options(-Wall -Wextra)
if(SV_TSAN)
  add_compile_options(-fsanitize=thread -g)
  add_link_options(-fsanitize=thread)
endif()
if(SV_ASAN)
  add_compile_options(-fsanitize=address,undefined -g)
  add_link_options(-fsanitize=address,undefined)
endif()

find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)

add_library(svcore STATIC src/s3client.cpp src/util.cpp)
target_include_directories(svcore PUBLIC src)
target_link_libraries(svcore PUBLIC OpenSSL::Crypto CURL::libcurl Threads::Threads)

add_executable(storvault-agent src/main.cpp)
target_link_libraries(storvault-agent PRIVATE svcore)

enable_testing()
add_executable(sv_tests tests/test_main.cpp)
target_link_libraries(sv_tests PRIVATE svcore)
add_test(NAME unit COMMAND sv_tests)

install(TARGETS storvault-agent RUNTIME DESTINATION bin)
